S.ai=S.ai||{};

const dgApplyAIBase=applyAI;
applyAI=function(out){
 dgApplyAIBase(out);
 if(out?.quota){S.ai.quota=out.quota;try{save()}catch{}}
 if(out?.servedBy){localStorage.setItem('dvirGymAiTestModel',String(out.servedBy))}
};

function dgAIErrorMessage(e){
 const s=String(e?.message||e||'');
 if(s.includes('openai_billing_required'))return 'המפתח תקין, אבל ל־OpenAI API אין כרגע יתרה/חיוב פעיל. הפעל Billing בחשבון ה־API ואז נסה שוב.';
 if(s.includes('no_supported_model'))return 'המפתח תקין, אבל לא נמצא מודל נתמך בחשבון ה־API הזה.';
 if(s.includes('openai_rate_limited'))return 'OpenAI מגביל כרגע בקשות. המתן מעט ונסה שוב.';
 if(s.includes('openai_key_rejected'))return 'OpenAI דחה את המפתח. ודא שה־API key פעיל ולא בוטל.';
 if(s.includes('invalid_key_format'))return 'פורמט המפתח לא נראה תקין.';
 if(s.includes('rate_limited'))return 'הגעת זמנית למכסת הבטיחות של המאמן. נסה שוב מאוחר יותר.';
 return 'לא הצלחתי להפעיל Cloud AI כרגע. Memory Engine ממשיך לעבוד, ואפשר לנסות שוב.';
}

dgActivateAI=async function(){
 const input=$('#dgOpenAIKey'),btn=$('#dgActivateAI');if(!input||!btn)return;
 const key=input.value.trim();if(key.length<30){toast('הדבק מפתח OpenAI API תקין');return}
 btn.disabled=true;btn.textContent='בודק מפתח + מודל + Billing…';
 try{
  const j=await dgKeyVault('set',key);input.value='';dgAiConfigured=true;localStorage.setItem('dvirGymAiConfigured','1');
  if(j.testedModel)localStorage.setItem('dvirGymAiTestModel',j.testedModel);
  toast(`✦ Cloud AI הופעל${j.testedModel?' · '+j.testedModel:''}`);renderMore();setTimeout(()=>{checkAIHealth();switchScreen('coach')},250)
 }catch(e){toast(dgAIErrorMessage(e));btn.disabled=false;btn.textContent='הפעל Cloud AI'}
};
